@RequestMapping(value = "/initNotFitAll", produces = MediaType.APPLICATION_JSON_VALUE)					
@ResponseBody					
public ResultVO<Map> calNotFillAll(HttpServletRequest request) {					
    Map params = MapConvert.arrayMapToStringMap(request.getParameterMap());					
    params.put("orderByClause", params.get("sortname") != null ? (params					
            .get("sortname").toString() + " " + params.get("sortorder")					
            .toString()) : "A.item_id");					
    // 录入查询参数					
    putQueryParams(params,request);					
    ResultVO<Map> rs = new ResultVO<Map>();					
    rs.setReturncode(DpConstant.OK);					
    // 执行父子路线检查					
    iqdRouteFitAnalyzeService.doParentChildRouteCheck(params);					
    if (DpConstant.ERROR.equals(params.get("returnCode").toString())) {					
        rs.setErrormessage(params.get("errorMessage") == null ? params					
                .get("sqlErrm").toString() : params.get("errorMessage")					
                .toString());					
        rs.setReturncode(DpConstant.ERROR);					
        return rs;					
    }					
    // 执行柳柳路线检查					
    iqdRouteFitAnalyzeService.doLiuLiuRouteCheck(params);					
    if (DpConstant.ERROR.equals(params.get("returnCode").toString())) {					
        rs.setErrormessage(params.get("errorMessage") == null ? params					
                .get("sqlErrm").toString() : params.get("errorMessage")					
                .toString());					
        rs.setReturncode(DpConstant.ERROR);					
        return rs;					
    }					
    rs.setModel(params);//回传会话ID					
    return rs;					
}

@RequestMapping("/saveAuditAgent")					
@ResponseBody					
public ResultVO<Map> saveAuditAgent(@RequestBody Map params, HttpServletRequest request) {					
    String sUserId = request.getSession().getAttribute("operaterid").toString();					
    String sUserName = request.getSession().getAttribute("operatername").toString();					
    params.put("I_SRC_AUDIT_UID", sUserId);					
    params.put("I_SRC_AUDIT_NAME", sUserName);					
    params.put("I_USER_ID", sUserId);					
    ResultVO<Map> rs = new ResultVO<Map>();					
    // 保存审核代理人					
    rs = iqdcustprddevjobService.saveAuditAgent(params);					
    return rs;					
}
