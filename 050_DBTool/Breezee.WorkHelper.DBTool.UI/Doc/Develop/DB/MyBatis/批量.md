## MyBatis-批量
### 批量新增
```
<insert id="insertUsers" parameterType="java.util.List">
    INSERT INTO users (id, name, email)
    VALUES
    <foreach collection="list" item="user" index="index" separator=",">
        (#{user.id}, #{user.name}, #{user.email})
    </foreach>
</insert>

<!--注意，使用batch元素进行批量插入时，需要在数据库连接字符串中添加rewriteBatchedStatements=true参数以启用JDBC批处理功能。-->
<insert id="batchInsert" parameterType="java.util.List">
    <foreach collection="list" item="item" batchSize="1000">
        insert into my_table (name, age) values (#{item.name}, #{item.age})
    </foreach>
</insert>

<!--属性维护-->			
<insert id="doCenterAttributeTmp" parameterType="java.util.List">			
    <foreach collection="list" item="item" separator=" " open="INSERT ALL" close="SELECT 1 FROM DUAL">			
       INTO IF_T_SEND_QDSAP_ITEM_TMP (prdmodel_code, ver_num, mfg_center_code)			
       VALUES  (#{item.PRDMODEL_CODE}, #{item.VER_NUM}, #{item.MFG_CENTER_CODE})			
    </foreach>			
</insert>	
```

### 批量修改
```
void auditApply(List<Map> params);		
<!--批量提交为待审核-->	
<update id="auditApply" parameterType="java.util.List">			
    BEGIN
    <foreach collection="list" item="item" index="index" separator=" ">	
        update MBOM_T_QDCUSTDEV_D t	
        set t.audit_flag='1',
        t.modifyer = #{item.USER_ID,jdbcType=VARCHAR},	
        t.modify_time = sysdate			
        where t.custdev_d_id = #{item.CUSTDEV_D_ID,jdbcType=VARCHAR};
    </foreach>
    END;
</update>

<update id="updateBatchMultiSql" parameterType="java.util.List">
    <foreach collection="list" item="item" separator=";">
        UPDATE user
        SET 
            username = #{item.username},
            age = #{item.age}
        WHERE id = #{item.id}
    </foreach>
</update>
```

### 批量删除
```
<!--批量删除指定项的变更单-->			
<update id="batchRemoveRouteHis">			
    delete from MBOM_T_QDITMROUTE_HIS A			
    where A.ROUTE_HIS_ID in (			
    <foreach collection="itemIdentifies" item="item" separator=",">			
        #{item.routeHisId}			
    </foreach>			
    )			
</update>	
```
