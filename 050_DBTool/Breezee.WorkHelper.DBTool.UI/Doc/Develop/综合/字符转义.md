## 字符转义
### XML转义
```
转义字符	符号	含义
&lt;	<	小于号
&gt;	>	大于号
&amp;	&	和
&apos;	'	单引号
&quot	"	双引号
```
### CDATA
在XML中，CDATA（Character Data）是一种特殊的数据类型，它允许我们在XML文档中插入文本，而不需要对其进行标记。  
格式：<![CDATA[ ... ]]>  
```
<![CDATA[  AND instr(#{lisType},b.COLCNCTYPE) > 0  ]]>
```

--前端查询：通过查询参数的方式，两次调用URL编码传入！！		
function queryCncReportQdRouteQuery(mfgCenterCode,cncCode,itemId){								
    if(cncCode==null || cncCode==''){								
       $.ligerDialog.warn('cncCode入参不能为空!');								
       return;								
    }								
    if(mfgCenterCode==null || mfgCenterCode==''){								
       $.ligerDialog.warn('mfgCenterCode入参不能为空!');								
       return;								
    }								
								
    var tabs = top.tab.getTabidList();								
    for(var i = 0 ; i < tabs.length ; i++){								
       if(tabs[i].indexOf("dialog_qdRouteQueryItem") > -1 ){								
          top.tab.removeTabItem(tabs[i]);								
       }								
    }								
    let urlAddr = "routequery/routequery_main_jump.action?cncCode="								
       +encodeURIComponent(encodeURIComponent(cncCode,"utf-8"),"utf-8")								
       +"&itemId="+encodeURIComponent(encodeURIComponent(itemId,"utf-8"),"utf-8")								
       +"&mfgCenterCode="+encodeURIComponent(encodeURIComponent(mfgCenterCode,"utf-8"),"utf-8")								
       +"&t="+new Date().getTime();								
    $.dp.common.createTab({								
       tabid:"dialog_qdRouteQueryItem",								
       title:"工艺路线查询",								
       url:urlAddr								
    });								
}

--后端跳转Controller代码：每个参数都要调用两次解码
@RequestMapping(value = "/routequery_main_jump")										
public String jumpIndex(HttpServletRequest request, ModelMap modelMap,										
                    @RequestParam(value = "cncCode", required = false) String cncCode,										
                    @RequestParam(value = "itemId", required = false) String itemId,										
                    @RequestParam(value = "mfgCenterCode", required = false) String mfgCenterCode){										
    iResourceRefService.curResourcRefinfo("dt_routedef", request, modelMap);										
    String sCncCode="";										
    String sItemId="";										
    String sMfgCenterCode="";										
    try{										
        sCncCode = URLDecoder.decode(cncCode, "UTF-8");	
        sCncCode = URLDecoder.decode(sCncCode, "UTF-8"); //每个参数都要调用两次解码										
        sItemId = URLDecoder.decode(itemId, "UTF-8");										
        sItemId = URLDecoder.decode(sItemId, "UTF-8");	//每个参数都要调用两次解码								
        sMfgCenterCode = URLDecoder.decode(mfgCenterCode, "UTF-8");										
        sMfgCenterCode = URLDecoder.decode(sMfgCenterCode, "UTF-8");										
    } catch (Exception e) {										
        e.printStackTrace();										
    }										
    //获取跳转过来的参数										
    if(sCncCode != null){										
        modelMap.put("jumpCncCode", sCncCode);	//参数回传给前端
    }										
    if(sItemId != null){										
        modelMap.put("jumpItemId", sItemId);										
    }										
    if(sMfgCenterCode != null){										
        modelMap.put("defaultMfgCenter", sMfgCenterCode);										
    }										
    return "mskbom/mbom/routequery/routequery_main";										
}
