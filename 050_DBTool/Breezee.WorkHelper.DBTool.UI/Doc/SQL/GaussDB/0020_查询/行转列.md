### 行转列
```
/*行转列*/
--原Oracle语句
Select i_sessionid As SESSION_ID,
        i_modeltag As MODELTAG,
        i_operaterid As OPER_ID,
        RN As DSEQ,
        Trim(D01),    
        Trim(D02),       
        Trim(D03),      
        Trim(D04)
From (Select ANALYZE_DATA, COL_ID, SEQID As RN
        From COM_T_ANALYZEDATA A
        Where A.SESSION_ID = i_sessionid
            And A.MODELTAG = i_modeltag
            And A.OPERATER_ID = i_operaterid)
PIVOT(Max(ANALYZE_DATA)
For COL_ID In(1 "D01", 2 "D02", 3 "D03", 4 "D04"));

--转换为GaussDB版本
--定义游标用于动态行转列
CURSOR c_import_data IS
SELECT SEQID AS RN,
	    MAX(CASE WHEN COL_ID = 1 THEN ANALYZE_DATA END) AS D01,
	    MAX(CASE WHEN COL_ID = 2 THEN ANALYZE_DATA END) AS D02,
	    MAX(CASE WHEN COL_ID = 3 THEN ANALYZE_DATA END) AS D03,
	    MAX(CASE WHEN COL_ID = 4 THEN ANALYZE_DATA END) AS D04,
	    MAX(CASE WHEN COL_ID = 5 THEN ANALYZE_DATA END) AS D05
FROM COM_T_ANALYZEDATA A
WHERE A.SESSION_ID = i_sessionid
	AND A.OPERATER_ID = i_operaterid
	AND A.MODELTAG = i_modletag
GROUP BY SEQID;

-- 使用CASE WHEN替代PIVOT进行行转列
FOR rec IN c_import_data LOOP
	INSERT INTO COM_T_DATAIMPORT
	    (SESSION_ID, MODELTAG, OPERATER_ID, DSEQ, D01, D02, D03, D04, D05)
	VALUES
	    (i_sessionid, i_modletag, i_operaterid, rec.RN, TRIM(rec.D01), TRIM(rec.D02), TRIM(rec.D03), TRIM(rec.D04), TRIM(rec.D05));
END LOOP;
```
