### 包示例
```
/*调用包示例：显式事务块（最可靠）*/	
BEGIN;	
DO $$	
DECLARE	
	o_Error_result SYS_REFCURSOR;	
	o_return_code VARCHAR;	
	o_error_message VARCHAR;	
	o_SQLERRM VARCHAR;	
	v_seq VARCHAR(400);	
	v_name VARCHAR(4000);	
BEGIN	
	ebom.pkg_jf_ebom_prdquery.P_ModulGroup_Imp(	
		'5F0B6700A574A5FB5918D230C361FBE5',	
		'5F0B6700A574A5FB5918D230C361FBE5',	
		'partGroup',	
		o_Error_result,	
		o_return_code,	
		o_error_message,	
		o_SQLERRM	
	);	
	-- 输出参数	
	RAISE NOTICE 'o_return_code: %', COALESCE(o_return_code, 'NULL');	
	RAISE NOTICE 'o_error_message: %', COALESCE(o_error_message, 'NULL');	
	RAISE NOTICE 'o_SQLERRM: %', COALESCE(o_SQLERRM, 'NULL');	
	-- 如果需要输出游标内容	
	IF o_Error_result IS NOT NULL THEN	
		LOOP	
			FETCH o_Error_result INTO v_seq, v_name;	
			EXIT WHEN o_Error_result%NOTFOUND;	
			RAISE NOTICE 'cursor data: id=%, name=%', v_seq, v_name;	
		END LOOP;	
		CLOSE o_Error_result;	
	END IF;	
END;	
$$;	
COMMIT;	


```
