## 性能优化
### Oracle游标更新方式修改为MERGE INTO更新方式
```
--取消游标的更新方式，修改为MERGE INTO更新方式，以下为Oralce示例：       
/*For r_up In (Select T.Rowid As row_id, A.ROUTE_MFGSTR As ROUTE_MFG_STR, A.ROUTE_ASSEMSTR As route_assem_str
                From ebom_t_qdexpand_tmp_down T
                Left Join (Select *
                          From MBOM_T_ROUTE_APPLY r
                        Where Not Exists (Select '1' From MBOM_T_ROUTE_APPLY h Where r.route_his_id = h.route_id)) A
                On A.VER_NUM = T.Ver_Num
                And A.ITEM_ID = T.ITEM_CID
                And a.mfg_center_code = i_mfgcentercode
            Where t.conversationid = i_ConversationId) Loop
      
Update ebom_t_qdexpand_tmp_down T
    Set route_mfg_str   = r_up.route_mfg_str, --【结构属性】-制造路线（编码）
        route_assem_str = r_up.route_assem_str --【结构属性】-装配路线（编码）
    Where Rowid = r_up.row_id;
      
End Loop;*/

MERGE INTO ebom_t_qdexpand_tmp_down U
USING (Select T.Rowid As row_id, A.ROUTE_MFGSTR As ROUTE_MFG_STR, A.ROUTE_ASSEMSTR As route_assem_str
        From ebom_t_qdexpand_tmp_down T
        Left Join (Select *
                    From MBOM_T_ROUTE_APPLY r
                    Where Not Exists (Select '1' From MBOM_T_ROUTE_APPLY h Where r.route_his_id = h.route_id)) A
            On A.VER_NUM = T.Ver_Num
        And A.ITEM_ID = T.ITEM_CID
        And a.mfg_center_code = i_mfgcentercode
        Where t.conversationid = i_ConversationId
    ) F
ON (U.Rowid = F.row_id)
    WHEN MATCHED THEN
    UPDATE
        SET route_mfg_str   = F.route_mfg_str, --【结构属性】-制造路线（编码）
            route_assem_str = F.route_assem_str; --【结构属性】-装配路线（编码）
```

### 函数作为查询条件时，可以多包一层，先查出函数结果，再过滤
```
--性能慢（40多秒才出来）
SELECT COUNT(*) AS total
FROM bom.EBOM_V_CNC A
LEFT JOIN bom.EBOM_T_CNC_ARCHIVE etca
       ON A.CNC_ID = etca.CNC_ID AND etca.CREATOR = '1'
LEFT JOIN public.t_eap_sys_user archiver
       ON etca.UPDATER = archiver.user_id
WHERE A.CNC_TYPE = 'dt_cnctype_eco_A5'
  AND bom.pkg_com_util.f_check_auth('1', A.CNC_ID, '') = 'Y';

--性能快（3秒就出来了）  
SELECT COUNT(*) AS total
FROM (
SELECT A.CNC_ID,bom.pkg_com_util.f_check_auth('1', A.CNC_ID, '') AS AUTH_FLAG
FROM bom.EBOM_V_CNC A
WHERE A.CNC_TYPE = 'dt_cnctype_eco_A5'
) A
WHERE AUTH_FLAG = 'Y';
```
### 查询列为调用包的函数，而函数是查询表，建议改左连接查询
```
--性能慢
select fawbom.pkg_com_operater.F_GetOpername(m.operid_pro) 
from t_test m
where m.test_id = '1';

--性能快
select P.user_name
from t_test m
LEFT JOIN public.t_eap_sys_user P ON P.user_id = m.operid_pro 
```
### 较多条件的exists可以考虑修改为join
```
--性能慢
Select to_char(t.timestamp_issue, 'yyyymmdd')
  Into var_vehdate
  From sys_t_cnc t
 Where Exists (Select 1
          From mbom_t_mfgitm a
         Where a.mfg_center_code = v_Mfg_Center_Code
           And a.item_id = v_prdmodel_code
           And a.date_bgn <= to_char(Sysdate, 'yyyymmdd')
         And a.date_end > to_char(Sysdate, 'yyyymmdd')
         And a.cncid_bgn = t.cnc_id);
--性能快
Select to_char(t.timestamp_issue, 'yyyymmdd')
          Into var_vehdate
          From sys_t_cnc t
          JOIN mbom_t_mfgitm a
              on a.mfg_center_code = v_Mfg_Center_Code
                   And a.item_id = v_prdmodel_code
                   And a.date_bgn <= to_char(Sysdate, 'yyyymmdd')
                   And a.date_end > to_char(Sysdate, 'yyyymmdd')
                   And a.cncid_bgn = t.cnc_id
         Where ROWNUM=1; 
```
