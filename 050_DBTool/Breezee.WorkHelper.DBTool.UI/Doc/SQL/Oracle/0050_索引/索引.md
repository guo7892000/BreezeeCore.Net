### 索引

/*增加索引*/
CREATE INDEX IDX01_T_UC_BU_BUY_CONTRACT ON T_UC_BU_BUY_CONTRACT(OPRATE_TIME);

/*增加唯一索引*/
CREATE UNIQUE INDEX UQ01_EMAIL_IDX ON EMPLOYEES(EMAIL);
CREATE UNIQUE INDEX UQ02_NAME_IDX ON EMPLOYEES(FIRST_NAME, LAST_NAME);
/*增加唯一索引*/
ALTER TABLE EMPLOYEES ADD CONSTRAINT UQ01_EMAIL UNIQUE (EMAIL);

DROP INDEX IDX01_NAME;
