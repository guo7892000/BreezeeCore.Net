### 锁
#### 查询被锁住的表
```
/*查询被锁住的表1*/
select a.OS_USER_NAME, c.owner, c.object_name, b.sid, b.serial#, logon_time
from v$locked_object a, v$session b, dba_objects c
where a.session_id = b.sid
and a.object_id = c.object_id
order by b.logon_time;

/*查询被锁住的表2*/
SELECT /*+ rule */ lpad(' ',decode(l.xidusn ,0,3,0))||l.oracle_username User_name, 
o.owner,o.object_name,o.object_type,s.sid,s.serial# 
FROM v$locked_object l,dba_objects o,v$session s 
WHERE l.object_id=o.object_id 
AND l.session_id=s.sid 
ORDER BY o.object_id,xidusn DESC;

```

#### 查询阻塞SQL（注：非死锁）
```
--查询阻塞SQL1（注：非死锁）
select final_blocking_instance, final_blocking_session, count(*)
from gv$session
where type = 'USER' 
and final_blocking_instance is not null
group by final_blocking_instance, final_blocking_session
order by 1 desc;
--查询阻塞SQL2（注：非死锁）
select distinct a.inst_id,a.sid,a.event,a.sql_id,a.prev_sql_id,a.status,a.username,a.program,a.type 
from gv$session a,
(select final_blocking_instance, final_blocking_session 
 from gv$session 
 where  final_blocking_instance is not null ) b
where a.inst_id=b.final_blocking_instance 
and a.sid=b.final_blocking_session 
order by 1;
```
